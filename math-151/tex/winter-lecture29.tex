\lecture{}

\begin{node}
The claim is the fundamental product theorem is equivalent to Bott
periodicity $\widetilde{K}(X)\iso\widetilde{K}(\Sigma^{2}X)$.
\end{node}

\subsection*{Step 1}

\begin{node}
Reduce $f$ to a Laurent polynomial in $z$, so we can assume
\begin{equation}
f_{z}(x)=\sum_{|k|\leq n}a_{k}(x)z^{k}
\end{equation}
where $a_{k}\colon E\to E$ is a bundle morphism.
\end{node}

\begin{proposition}
Any vector bundle $[E,f]$ is isomorphic to a vector bundle of the form
$[E,f']$ where
\begin{equation}
f'_{z}(x)=\sum_{|k|\leq n}a_{k}(x)z^{k}.
\end{equation}
\end{proposition}
