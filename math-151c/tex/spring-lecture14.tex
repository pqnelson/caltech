\lecture{}

\begin{theorem}
Let $(M,g)$ be a Riemannian manifold, let $\gamma\colon[a,b]\to M$ be
a geodesic. Then for every $t_{0}\in[a,b]$ there exists an
$\varepsilon>0$ such that for every $t\in(t_{0},t_{0}+\varepsilon]$ we have
\begin{equation}
\D_{g}(\gamma(t),\gamma(t_{0}))=\length_{g}(\gamma|_{[t_{0},t]}).
\end{equation}
\end{theorem}
