\subsection{Topological Spaces}

\begin{definition}
Let $X$ be a set. Let $\powerset{X}$ be the powerset of $X$ (i.e., the
set of all subsets of $X$). The subset $\topology\subset\powerset{X}$ is
called a \define{Topology} of $X$ if
\begin{enumerate}
\item $\emptyset\in\topology$ and $X\in\topology$
\item $\topology$ is closed under finite intersections $U_{1}\cap\cdots\cap U_{n}\in\topology$
  where $U_{j}\in\topology$ for $j=1,\dots,n$
\item $\topology$ is closed under arbitrary unions $\bigcup_{i\in I}U_{i}\in\topology$
  for any family $\{U_{i}\in\topology\}_{i\in I}$ with indexing set $I$.
\end{enumerate}
The pair $(X,\topology)$ of a set with a topology on it constitutes a
\define{Topological Space}
\end{definition}

\begin{definition}
Let $(X,\topology_{X})$ and $(Y,\topology_{Y})$ be topological spaces.
We say they are \define{Topologically Equivalent} if there exists a
bijection $f\colon X\to Y$ such that for all $U$ we have
$U\in\topology_{X}\iff f(U)\in\topology_{Y}$. In this case, we write
$(X,\topology_{X})\TopEquiv(Y,\topology_{Y})$ to indicate the spaces are
topologically equivalent.
\end{definition}

\begin{example}
The trivial topology $\topology=\{\emptyset,X\}$ is the smallest topology
and it is the weakest topology.
\end{example}

\begin{example}
The strongest topology on $X$ is $\powerset{X}=2^{X}$, and it is
called the \define{Discrete Topology}.
\end{example}

\begin{example}
The \define{Cofinite Topology} on $X$ is the set $\topology$ consisting of
complements of finite subsets of $X$. (We can also generalize this to
the \define{Co-countable Topology} consisting of complements of
countable subsets of $X$.)
\end{example}

\begin{definition}
Let $X$ be a set, let $x\in X$ and let $U\subset X$.
We say $U$ is a \define{Neighborhood} of $x$ if it belongs to the
interior of $U$, i.e., $x\in\Interior(U)$.
\end{definition}

\begin{definition}
Let $(X,\topology)$ be topological spaces. Let $E\subset X$ be a subset.
The \define{Relative Topology} on $E$ is $\topology_{E}=\{U\cap E\mid U\in\topology\}$.
We need to prove $\topology_{E}$ is a topology on $E$, but the verification
is straightforward.
\end{definition}

\begin{example}
Let $X=\RR$ equipped with the standard topology. Let $E=\ZZ$. We claim
the relative topology $\topology_{E}$ is discrete.
\end{example}

\begin{proof}
  Suffices to prove every point is open. We see
  \begin{equation}
\{0\}=\ZZ\cap(-1/2,1/2)
  \end{equation}
  is open in $\topology_{E}$.
\end{proof}

\begin{definition}
Let $(X,\rho)$ be a metric space. Recall $B(x,r)=\{y\in X\mid\rho(x,y)<r\}$
is the open ball of radius $r$ centered at $x\in X$. Then we define
the \define{Metric Topology} $\topology(\rho)$ on $X$ to consist of all
$U\in\powerset{X}$ such that for all $x\in U$ there exists a radius
$r>0$ such that $B(x,r)\subset U$.
\end{definition}

\begin{definition}
Let $X$ be a set equipped with an order $>$.
The \define{Order Topology} is generated by sets of the form
$U_{a}=\{x\in X\mid x>a\}$ and $U'_{a}=\{x\in X\mid x<a\}$.
\end{definition}

\begin{node}
The standard topology on $\RR$ is given by the metric topology using
the metric $d(x,y)=|x-y|$.
\end{node}

\begin{definition}
Let $\topology$ be a topology on $X$. Let $\beta\subset\topology$ be some
subset. Then we say $\beta$ is a \define{Base} of $\topology$ if every open
set $U\in\topology$ is the union of some subfamily of $\beta$.
\end{definition}

\subsection{Topology generated by a family of subsets}

\begin{definition}
Let $X$ be a set. Let $\beta\subset\powerset{X}$ be some family of
subsets of $X$. Then we can define $\topology(\beta)$ as the weakest
topology of $X$ containing $\beta\subset\topology(\beta)$ all elements of $\beta$.
\end{definition}

\begin{theorem}
Finite intersections of sets in $\beta$ is a base of the topology
$\topology(\beta)$ generated by $\beta$.
\end{theorem}
(This gives another description of the topology generated by the
family $\beta$ of subsets of $X$.)

\begin{example}
Consider $\beta=\{(a,b]\subset\RR\}$ all half-open intervals of the
real line. We claim $\beta$ is a base of $\topology(\beta)$. The problem
is to show the closure of $(a,b]$ with respect to $\topology(\beta)$ is
$(a,b]$. So these intervals are simultaneously open \emph{and}
closed (with respect to this topology).
Then $(\RR,\topology(\beta))$ is disconnected.
\end{example}

\begin{theorem}
Recall, a topological space $(X,\topology)$ is connected if and only if the
only subsets which are both open and closed are $X$ and
$\emptyset$. In particular, if there is a proper nonempty subset
$U\propersubset X$ which is both open and closed, then $(X,\topology)$ is
disconnected. 
\end{theorem}

(Some authors do not consider the empty topological space
$(\emptyset,\topology)$ to be connected, but it really does not matter. We
consider it connected.)
